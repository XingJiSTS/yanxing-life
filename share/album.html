async function loadAlbumImages(){ const params = new URLSearchParams(location.search); const albumId = params.get("albumId"); // 增加友好提示，不再纯黑底白字 if(!albumId){ document.body.style.background = "#f7f9fc"; document.body.innerHTML = `
该页面仅能通过相册二维码/图库相册图片打开

返回主页
`; return; } try { const res = await fetch(`${WORKER_URL}list`); const data = await res.json(); const allFiles = data.files || []; imgList = allFiles.filter(file => file.albumId === albumId).sort((a,b)=>a.time - b.time); if(imgList.length === 0){ document.body.style.background = "#f7f9fc"; document.body.innerHTML = `
当前相册无图片

返回主页
`; return; } renderImages(); bindTouch(); updatePage(); } catch(err){ document.body.style.background = "#f7f9fc"; document.body.innerHTML = `
相册加载失败，请刷新页面

返回主页
`; console.error(err); } }
